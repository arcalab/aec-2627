\begin{frame}{The rule pattern}

        \[
                \infer{
                        \text{conclusion}
                }{
                        \text{assumption}_1 
                        \quad \dots \quad
                        \text{assumption}_n
                }
        \]

        \pause
        \vspace{1cm}
        A rule for each program construct

        + a logical rule for changing pre- and post-conditions
\end{frame}

\begin{frame}{The proof pattern}

        Program's \salert{syntactic structure} determines which rule to apply

        Developing a correctness proof $\Rightarrow$ making a \salert{tree grow}
        
        \vspace{0.5cm}
        \[
                \infer{
                        \tikzmarknode{concl}{\text{conclusion}}
                }{
                        \infer{
                                \text{assumption} 
                        }{
                                \infer{
                                \text{assumption}
                                }{ 
                                        \cdots
                                }
                                \, \cdots \,
                                \infer{
                                \text{assumption}
                                }{
                                        \checkmark
                                }
                        }
                        \quad \cdots \quad
                        \infer{
                                \text{assumption}
                        }{\cdots}
                }
        \]

        \begin{tikzpicture}[remember picture,overlay]
                \draw[<-,thick,red] (concl.south)+(0.0,-0.2)  -- ++(0.0,-0.2)
                        -- ++(0.0,-0.3)

        \end{tikzpicture}

\end{frame}

\newcommand{\blue}[1]{\textcolor{blue}{#1}}
\begin{frame}{The rules}
        \[
                \infer{\{ \Phi \} \> \prog{skip} \> \{ \Phi \} }{}
                \hspace{3cm}
                \infer{\{ \Phi[\prog{e}/\prog{x}] \} \> \prog{ x : = e } \>
                \{ \Phi \} }{}
        \]
        \\[1pt]
        \[
                \infer { \{ \Phi \} \> \prog{p \> \blue{ ;} \> q} \> \{ \Xi \} }{
                        \{ \Phi \} \> \prog{p} \> \{ \Psi \}
                        \qquad
                        \{ \Psi \} \> \prog{q} \> \{ \Xi \}
                }
                \hspace{3cm}
                \infer{
                        \{ \Phi \} \> \prog{\blue{while}} \> \prog{b} \>
                        \prog{p} \> 
                        \{ \neg \prog{b} \wedge \Phi \}
                }{
                        \{ \Phi \wedge \prog{b} \} \> \prog{p} \> \{ \Phi \}
                }
        \]
        \\[1pt]
        \[
                \infer {
                        \{ \Phi \} \> \prog{\blue{if} \> b}
                        \> \prog{\blue{then}} \>
                        \prog{p} \> \prog{\blue{else}} \> \prog{q}
                        \> \{ \Psi \}
                }{
                        \{ \prog{b} \wedge \Phi \} \> \prog{p} \> \{ \Psi \}
                        \qquad
                        \{ \prog{\neg b} \wedge \Phi \} \> \prog{q} \> \{ \Psi \}
                }
        \]
        \\[1pt]
        \[
                \infer{ \{ \Phi \} \> \prog{p} \> \{ \Omega \} }{
                         \Phi \rightarrow \Psi
                        \quad
                        \{ \Psi \} \> \prog{p} \> \{ \Xi \}
                        \quad
                        \Xi \rightarrow \Omega
                }
        \]
\end{frame}
