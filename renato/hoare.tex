\section{First-order logic (FOL)}

\begin{frame}{Our old friend, first-order logic, to the rescue}

        Rigorous

        Deceptively simple

        Powerful enough to handle a large portion of \salert{mathematics}
\end{frame}


\begin{frame}{Basic ingredients}

        \begin{itemize}
                \item Variables (\ie\ placeholders) $x : \typeA$

                \item Operations (\eg\ addition) $f : \typeA_1, \dots, \typeA_n \to \typeA$

                \item Predicates (\eg\ equality) $p : \typeA_1, \dots, \typeA_n$

                \item Connectives $\wedge, \Rightarrow, \neg$

                \item Quantifiers $\forall x : \typeA. \, \Phi(x), 
                        \, \exists x : \typeA. \, \Phi(x)$
        \end{itemize}
        \pause
        \vspace{0.2cm}
        \dots\ plus a set of rules that underlie \salert{human reasoning} and
        \salert{advanced PLs} -- and which hopefully you will see in later
        courses ;-)
\end{frame}

\begin{frame}{From natural language to FOL}
        Let's translate the following sentences to FOL (\ie\ first-order logic)
        %
        \begin{enumerate}
                \item $m$ is an upper bound of $x$ and $y$
                \item $m$ is the maximum of $x$ and $y$
                \item all natural numbers are either odd or even
                \item there exists no real number that is equal
                        to the square root of $-1$
                \item $i$ is the largest number for which predicate $\phi : \Nats$
                        holds
                \item every nat. nº greater than $2$ is the sum
                        of two prime nºs  (\salert{Golbach's conjecture})
        \end{enumerate}
\end{frame}

\begin{frame}{A word of Warning -- Specifications vs. Implementations}

        FOL is a logic for reasoning about  \dots\  properties
        (\salert{specifications})

        A programming language is a language for \dots\  programming
        (\salert{implementations})

        Don't try to program in FOL nor to specify in \texttt{C}

\end{frame}

\begin{frame}{The rule pattern}

        \[
                \infer{
                        \text{conclusion}
                }{
                        \text{condition}_1 
                        \quad \dots \quad
                        \text{condition}_n
                }
        \]
\end{frame}

\newcommand{\blue}[1]{\textcolor{blue}{#1}}
\begin{frame}{The rules}
        \[
                \infer{\{ \Phi \} \> \prog{skip} \> \{ \Phi \} }{}
                \hspace{3cm}
                \infer{\{ \Phi[\prog{e}/\prog{x}] \} \> \prog{ x : = e } \>
                \{ \Phi \} }{}
        \]
        \\[1pt]
        \[
                \infer { \{ \Phi \} \> \prog{p \> \blue{ ;} \> q} \> \{ \Xi \} }{
                        \{ \Phi \} \> \prog{p} \> \{ \Psi \}
                        \qquad
                        \{ \Psi \} \> \prog{q} \> \{ \Xi \}
                }
                \hspace{3cm}
                \infer{
                        \{ \Phi \} \> \prog{\blue{while}} \> \prog{b} \>
                        \prog{p} \> 
                        \{ \neg \prog{b} \wedge \Phi \}
                }{
                        \{ \Phi \wedge \prog{b} \} \> \prog{p} \> \{ \Phi \}
                }
        \]
        \\[1pt]
        \[
                \infer {
                        \{ \Phi \} \> \prog{\blue{if} \> b}
                        \> \prog{\blue{then}} \>
                        \prog{p} \> \prog{\blue{else}} \> \prog{q}
                        \> \{ \Psi \}
                }{
                        \{ \prog{b} \wedge \Phi \} \> \prog{p} \> \{ \Psi \}
                        \qquad
                        \{ \prog{\neg b} \wedge \Phi \} \> \prog{q} \> \{ \Psi \}
                }
        \]
        \\[1pt]
        \[
                \infer{ \{ \Phi \} \> \prog{p} \> \{ \Psi \} }{
                        \vdash \Phi \rightarrow \Psi
                        \quad
                        \{ \Psi \} \> \prog{p} \> \{ \Xi \}
                        \quad
                        \vdash \Xi \rightarrow \Omega
                }
        \]
\end{frame}
